###########################################################################
# Borderlands Rising - on actions
#
# A single startup hook, guarded by a global flag so that it runs once.
# The only recurring work in the mod is done by self-scheduling monthly
# events (Palanmir's data collection and the Kurdish unrest roll), never
# by daily or weekly pulses.
###########################################################################

on_actions = {
	on_startup = {
		effect = {
			if = {
				limit = { NOT = { has_global_flag = BRS_startup_done } }
				set_global_flag = BRS_startup_done

				# Diplomacy and balance of power that the base game only ever
				# sets up at game start, not in its 1936 history files.
				if = {
					limit = { country_exists = DNZ }
					# Poland's history file marks Danzig as an "international
					# city" (-100% recruitable population and local factories),
					# which would cripple a Free City that owns only Danzig.
					85 = {
						remove_dynamic_modifier = { modifier = international_city }
					}
					DNZ = {
						if = {
							limit = { has_power_balance = { id = DZG_danzig_question } }
							add_power_balance_modifier = {
								id = DZG_danzig_question
								modifier = DZG_bop_forster_agitation
							}
						}
					}
				}
				# The armistice of 1920 between South Russia and the Soviets.
				if = {
					limit = {
						country_exists = CRI
						country_exists = SOV
					}
					CRI = {
						diplomatic_relation = {
							country = SOV
							relation = non_aggression_pact
							active = yes
						}
					}
				}
				# The Treaty of Moscow (1920) between Azerbaijan and the Soviets.
				if = {
					limit = {
						country_exists = AZR
						country_exists = SOV
					}
					AZR = {
						diplomatic_relation = {
							country = SOV
							relation = non_aggression_pact
							active = yes
						}
					}
				}

				# Garrisons of the old owners leave the new countries.
				85 = {
					teleport_armies = {
						limit = { tag = POL }
						to_state = 807
					}
					teleport_armies = {
						limit = { tag = GER }
						to_state = 5
					}
				}
				137 = {
					teleport_armies = {
						limit = { tag = SOV }
						to_state = 196
					}
				}
				229 = {
					teleport_armies = {
						limit = { tag = SOV }
						to_state = 232
					}
				}

				# Lithuania's focus tree (game rule). The Palanmir tree is picked at
				# load; the vanilla option loads the base game's tree instead.
				if = {
					limit = {
						country_exists = LIT
						has_game_rule = { rule = BRS_lithuania_tree option = VANILLA }
					}
					LIT = {
						if = {
							limit = { has_dlc = "No Step Back" }
							load_focus_tree = { tree = lithuania_tree keep_completed = no }
							else = {
								load_focus_tree = { tree = generic_focus keep_completed = no }
							}
						}
					}
				}

				# The Kurds of Turkey and Iraq (game rule "Kurdish resistance").
				if = {
					limit = {
						NOT = { has_game_rule = { rule = BRS_kurdish_resistance option = VANILLA } }
					}
					KRD_setup_kurdish_resistance = yes
				}
			}
		}
	}
}
