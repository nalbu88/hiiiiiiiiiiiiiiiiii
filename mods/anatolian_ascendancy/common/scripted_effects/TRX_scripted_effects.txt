# Unlocks the Janissary Guard battalion and creates its template once.
# Reachable from both the army branch and the Ottoman branch.
TRX_unlock_janissary_guard = {
	unlock_subunit = trx_janissary_guard
	custom_effect_tooltip = TRX_unlock_janissary_tt
	if = {
		limit = { NOT = { has_country_flag = TRX_janissary_template_created } }
		set_country_flag = TRX_janissary_template_created
		hidden_effect = {
			division_template = {
				name = "Yeniçeri Muhafız Tümeni"
				regiments = {
					trx_janissary_guard = { x = 0 y = 0 }
					trx_janissary_guard = { x = 0 y = 1 }
					trx_janissary_guard = { x = 0 y = 2 }
					trx_janissary_guard = { x = 1 y = 0 }
					trx_janissary_guard = { x = 1 y = 1 }
					trx_janissary_guard = { x = 1 y = 2 }
					artillery_brigade = { x = 2 y = 0 }
					artillery_brigade = { x = 2 y = 1 }
				}
			}
		}
	}
}

# Picks the political path the AI will follow, based on the game rule.
# Runs once at game start, only for an AI-controlled Turkey.
TRX_set_ai_path = {
	if = {
		limit = { has_game_rule = { rule = TRX_ai_behavior option = KEMALIST } }
		set_country_flag = TRX_ai_kemalist
	}
	else_if = {
		limit = { has_game_rule = { rule = TRX_ai_behavior option = DEMOCRATIC } }
		set_country_flag = TRX_ai_democratic
	}
	else_if = {
		limit = { has_game_rule = { rule = TRX_ai_behavior option = OTTOMAN } }
		set_country_flag = TRX_ai_ottoman
	}
	else_if = {
		limit = { has_game_rule = { rule = TRX_ai_behavior option = TURANIST } }
		set_country_flag = TRX_ai_turanist
	}
	else_if = {
		limit = { has_game_rule = { rule = TRX_ai_behavior option = SOCIALIST } }
		set_country_flag = TRX_ai_socialist
	}
	else_if = {
		limit = { has_game_rule = { rule = TRX_ai_behavior option = RANDOM } }
		random_list = {
			30 = { set_country_flag = TRX_ai_kemalist }
			20 = { set_country_flag = TRX_ai_democratic }
			20 = { set_country_flag = TRX_ai_ottoman }
			15 = { set_country_flag = TRX_ai_turanist }
			15 = { set_country_flag = TRX_ai_socialist }
		}
	}
	else = {
		# DEFAULT: historical when historical focus is on, mostly Kemalist otherwise
		if = {
			limit = { is_historical_focus_on = yes }
			set_country_flag = TRX_ai_kemalist
		}
		else = {
			random_list = {
				45 = { set_country_flag = TRX_ai_kemalist }
				15 = { set_country_flag = TRX_ai_democratic }
				15 = { set_country_flag = TRX_ai_ottoman }
				15 = { set_country_flag = TRX_ai_turanist }
				10 = { set_country_flag = TRX_ai_socialist }
			}
		}
	}
}
