# Picks the political path the AI will follow, based on the game rule.
# Runs once at game start, only for an AI-controlled Turkey.
TRX_set_ai_path = {
	if = {
		limit = { has_game_rule = { rule = TRX_ai_behavior option = KEMALIST } }
		set_country_flag = TRX_ai_kemalist
	}
	else_if = {
		limit = { has_game_rule = { rule = TRX_ai_behavior option = DEMOCRATIC } }
		set_country_flag = TRX_ai_democratic
	}
	else_if = {
		limit = { has_game_rule = { rule = TRX_ai_behavior option = OTTOMAN } }
		set_country_flag = TRX_ai_ottoman
	}
	else_if = {
		limit = { has_game_rule = { rule = TRX_ai_behavior option = TURANIST } }
		set_country_flag = TRX_ai_turanist
	}
	else_if = {
		limit = { has_game_rule = { rule = TRX_ai_behavior option = SOCIALIST } }
		set_country_flag = TRX_ai_socialist
	}
	else_if = {
		limit = { has_game_rule = { rule = TRX_ai_behavior option = RANDOM } }
		random_list = {
			30 = { set_country_flag = TRX_ai_kemalist }
			20 = { set_country_flag = TRX_ai_democratic }
			20 = { set_country_flag = TRX_ai_ottoman }
			15 = { set_country_flag = TRX_ai_turanist }
			15 = { set_country_flag = TRX_ai_socialist }
		}
	}
	else = {
		# DEFAULT: historical when historical focus is on, mostly Kemalist otherwise
		if = {
			limit = { is_historical_focus_on = yes }
			set_country_flag = TRX_ai_kemalist
		}
		else = {
			random_list = {
				45 = { set_country_flag = TRX_ai_kemalist }
				15 = { set_country_flag = TRX_ai_democratic }
				15 = { set_country_flag = TRX_ai_ottoman }
				15 = { set_country_flag = TRX_ai_turanist }
				10 = { set_country_flag = TRX_ai_socialist }
			}
		}
	}
}

###########################################################################
# Kemalism: the Six Arrows
# Every second arrow upgrades the spirit; all six make it the Kemalist State.
###########################################################################

TRX_add_arrow = {
	custom_effect_tooltip = TRX_add_arrow_tt
	hidden_effect = {
		add_to_variable = { TRX_arrows = 1 }
		if = {
			limit = {
				has_idea = TRX_six_arrows_1
				check_variable = { TRX_arrows > 1 }
			}
			swap_ideas = { remove_idea = TRX_six_arrows_1 add_idea = TRX_six_arrows_2 }
		}
		else_if = {
			limit = {
				has_idea = TRX_six_arrows_2
				check_variable = { TRX_arrows > 3 }
			}
			swap_ideas = { remove_idea = TRX_six_arrows_2 add_idea = TRX_six_arrows_3 }
		}
		else_if = {
			limit = {
				has_idea = TRX_six_arrows_3
				check_variable = { TRX_arrows > 5 }
			}
			swap_ideas = { remove_idea = TRX_six_arrows_3 add_idea = TRX_kemalist_state }
		}
	}
}

###########################################################################
# Factions
# Non-aligned governments may not be allowed to found factions. The rule
# is lifted for the instant the faction is created, then restored.
###########################################################################

TRX_allow_faction_creation = {
	if = {
		limit = { NOT = { has_rule = can_create_factions } }
		set_country_flag = TRX_temp_faction_rule
		set_rule = { can_create_factions = yes }
	}
}

TRX_restore_faction_rule = {
	if = {
		limit = { has_country_flag = TRX_temp_faction_rule }
		clear_rule = { can_create_factions = yes }
		clr_country_flag = TRX_temp_faction_rule
	}
}
