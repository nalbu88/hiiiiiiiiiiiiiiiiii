###########################################################################
# Anatolian Ascendancy - country events
#
# Every event is "is_triggered_only": nothing here is polled by the
# mean-time-to-happen system, so the events cost nothing until a focus,
# decision or another event fires them.
#
# In events sent by Turkey to another country, ROOT is the receiver and
# FROM is Turkey.
###########################################################################

add_namespace = trx

	}
}

### Eastern Question: the Dersim crisis boils over (mission timeout) ###
country_event = {
	id = trx.70
	title = trx.70.t
	desc = trx.70.d
	picture = GFX_report_event_dead_soldiers
	is_triggered_only = yes

	option = {
		name = trx.70.a
		ai_chance = { base = 50 }
		if = {
			limit = { has_idea = TRX_eastern_question }
			swap_ideas = {
				remove_idea = TRX_eastern_question
				add_idea = TRX_eastern_pacification
			}
		}
		set_country_flag = TRX_eastern_policy_chosen
		add_stability = -0.05
	}
	option = {
		name = trx.70.b
		ai_chance = { base = 50 }
		if = {
			limit = { has_idea = TRX_eastern_question }
			swap_ideas = {
				remove_idea = TRX_eastern_question
				add_idea = TRX_eastern_reconciliation
			}
		}
		set_country_flag = TRX_eastern_policy_chosen
		add_political_power = -100
	}
}
