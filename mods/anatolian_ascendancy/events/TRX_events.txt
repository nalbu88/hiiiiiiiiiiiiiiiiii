###########################################################################
# Anatolian Ascendancy - country events
#
# Every event is "is_triggered_only": nothing here is polled by the
# mean-time-to-happen system, so the events cost nothing until a focus,
# decision or another event fires them.
#
# In events sent by Turkey to another country, ROOT is the receiver and
# FROM is Turkey.
###########################################################################

### Communist path: Turkey asks to join the Comintern (sent to the Soviet Union) ###

	id = trx.90
	title = trx.90.t
	desc = trx.90.d
	picture = GFX_report_event_generic_handshake
	is_triggered_only = yes

	option = {
		name = trx.90.a
		ai_chance = {
			base = 90
			modifier = { factor = 0 has_war_with = FROM }
		}
		trigger = {
			OR = {
				is_faction_leader = yes
				is_in_faction = no
			}
		}
		if = {
			limit = { is_in_faction = no }
			TRX_allow_faction_creation = yes
			create_faction_from_template = {
				template = TRX_faction_comintern
				name = TRX_faction_comintern_name
			}
			TRX_restore_faction_rule = yes
		}
		add_to_faction = FROM
		FROM = { country_event = { id = trx.91 hours = 6 } }
	}
	option = {
		name = trx.90.b
		ai_chance = { base = 10 }
		FROM = { country_event = { id = trx.92 hours = 6 } }
	}
}
